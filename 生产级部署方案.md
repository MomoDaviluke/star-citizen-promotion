#### 任务 1.1：JWT 存储安全改造（1 天）

**问题**：当前 JWT 存储在 localStorage，存在 XSS 攻击风险

**解决方案**：迁移至 httpOnly Cookie + CSRF 防护

**执行步骤：**

**Step 1：后端改造（`server/src/utils/jwt.js`）**
```javascript
// 原代码（需要修改）
// 登录成功后返回 { success: true, data: { user, token } }
// 前端将 token 存储到 localStorage

// 新方案：使用 httpOnly Cookie
import { serialize } from 'cookie'

export function setTokenCookie(res, token) {
  const cookieOptions = {
    httpOnly: true,        // 防止 XSS 读取
    secure: process.env.NODE_ENV === 'production',  // HTTPS 下必须
    sameSite: 'strict',    // 防止 CSRF
    maxAge: 7 * 24 * 60 * 60 * 1000,  // 7 天
    path: '/'
  }
  
  res.setHeader('Set-Cookie', serialize('token', token, cookieOptions))
}

// 从 Cookie 中读取 token
export function getTokenFromCookie(req) {
  const token = req.cookies?.token
  return token || null
}
```

**Step 2：修改登录/注册接口（`server/src/routes/auth.js`）**
```javascript
// 登录接口改造
router.post('/login', async (req, res, next) => {
  try {
    const { email, password } = req.body
    const result = await authService.loginUser(email, password)
    
    // 设置 httpOnly Cookie
    setTokenCookie(res, result.token)
    
    // 返回用户信息（不包含 token）
    res.json({
      success: true,
      data: { user: result.user }
    })
  } catch (error) {
    next(error)
  }
})
```

**Step 3：前端改造（`src/services/auth.js`）**
```javascript
// 移除 localStorage 存储 token 的逻辑
// 登录请求后，不需要手动存储 token（由 Cookie 自动管理）

export async function login(credentials) {
  const response = await httpClient.post('/api/auth/login', credentials)
  
  // 不再存储 token 到 localStorage
  // 只存储用户信息
  if (response.data.success) {
    localStorage.setItem('user', JSON.stringify(response.data.data.user))
  }
  
  return response.data
}

// 请求拦截器改造（src/services/http.js）
httpClient.interceptors.request.use(config => {
  // 不再从 localStorage 读取 token
  // Cookie 会自动随请求发送
  return config
})
```

**Step 4：修改认证中间件（`server/src/middleware/auth.js`）**
```javascript
import { getTokenFromCookie } from '../utils/jwt.js'

export function authenticate(req, res, next) {
  // 从 Cookie 中读取 token
  const token = getTokenFromCookie(req)
  
  if (!token) {
    return res.status(401).json({
      success: false,
      error: { code: 'UNAUTHORIZED', message: '未提供认证令牌' }
    })
  }
  
  try {
    const decoded = jwt.verify(token, process.env.JWT_SECRET)
    req.user = decoded
    next()
  } catch (error) {
    return res.status(401).json({
      success: false,
      error: { code: 'UNAUTHORIZED', message: '无效的认证令牌' }
    })
  }
}
```

```javascript
import { validationResult } from 'express-validator'

export function validate(validations) {
  return async (req, res, next) => {
    // 执行所有验证
    await Promise.all(validations.map(validation => validation.run(req)))
    
    const errors = validationResult(req)
    if (errors.isEmpty()) {
      return next()
    }
    
    // 返回验证错误
    return res.status(400).json({
      success: false,
      error: {
        code: 'VALIDATION_ERROR',
        message: '请求参数错误',
        details: errors.array()
      }
    })
  }
}
```

**Step 3：定义验证规则（`server/src/validators/authValidator.js`）**

**Step 2：创建清理脚本（`server/src/database/cleanupAuditLogs.js`）**
```javascript
import { query, execute } from './pool.js'
import { config } from '../config/index.js'

export async function cleanupAuditLogs() {
  const { retentionDays, batchSize } = config.auditLog
  const cutoffDate = new Date()
  cutoffDate.setDate(cutoffDate.getDate() - retentionDays)
  
  console.log(`🧹 清理 ${retentionDays} 天前的审计日志（截止 ${cutoffDate.toISOString()}）`)
  
  let totalDeleted = 0
  let deleted = 0
  
  do {
    // 批量删除（避免锁表太久）
    const result = await execute(
      'DELETE FROM activity_logs WHERE created_at < ? LIMIT ?',
      [cutoffDate, batchSize]
    )
    
    deleted = result.affectedRows
    totalDeleted += deleted
    
    console.log(`  删除 ${deleted} 条记录`)
  } while (deleted === batchSize)
  
  console.log(`✅ 清理完成，共删除 ${totalDeleted} 条记录`)
  
  return totalDeleted
}
```

**Step 3：定时任务（`server/src/tasks/scheduler.js`）**
```javascript
import { cleanupAuditLogs } from '../database/cleanupAuditLogs.js'

export function startScheduler() {
  // 每天凌晨 2 点执行清理
  const TWENTY_FOUR_HOURS = 24 * 60 * 60 * 1000
  
  setInterval(async () => {
    const now = new Date()
    if (now.getHours() === 2 && now.getMinutes() === 0) {
      try {
        await cleanupAuditLogs()
      } catch (error) {
        console.error('❌ 审计日志清理失败:', error)
      }
    }
  }, 60 * 1000)  // 每分钟检查
  
  console.log('✅ 定时任务已启动（审计日志清理）')
}
```

```typescript
import { transaction } from '../database/pool.js'

export async function registerUser(username, email, password) {
  return await transaction(async (connection) => {
    // 检查用户名是否已存在
    const [existingUsers] = await connection.execute(
      'SELECT id FROM users WHERE username = ? OR email = ?',
      [username, email]
    )
    
    if (existingUsers.length > 0) {
      throw new ConflictError('用户名或邮箱已存在')
    }
    
    // 创建用户
    const userId = uuidv4()
    const passwordHash = await bcrypt.hash(password, 12)
    
    await connection.execute(
      'INSERT INTO users (id, username, email, password_hash, role) VALUES (?, ?, ?, ?, ?)',
      [userId, username, email, passwordHash, 'member']
    )
    
    // 同时创建成员记录
    await connection.execute(
      'INSERT INTO members (id, name, status) VALUES (?, ?, ?)',
      [userId, username, 'active']
    )
    
    // 记录审计日志
    await connection.execute(
      'INSERT INTO activity_logs (user_id, action, entity, entity_id) VALUES (?, ?, ?, ?)',
      [userId, 'register', 'user', userId]
    )
    
    // 返回用户信息（不包含密码）
    const [users] = await connection.execute(
      'SELECT id, username, email, role FROM users WHERE id = ?',
      [userId]
    )
    
    return users[0]
  })
}
```

```typescript
export async function updateApplicationStatus(applicationId, status, reviewerId) {
  return await transaction(async (connection) => {
    // 检查申请是否存在
    const [applications] = await connection.execute(
      'SELECT * FROM applications WHERE id = ?',
      [applicationId]
    )
    
    if (applications.length === 0) {
      throw new NotFoundError('申请不存在')
    }
    
    // 更新申请状态
    await connection.execute(
      'UPDATE applications SET status = ?, reviewed_by = ?, reviewed_at = NOW() WHERE id = ?',
      [status, reviewerId, applicationId]
    )
    
    // 如果批准，创建成员记录
    if (status === 'approved') {
      const app = applications[0]
      
      await connection.execute(
        'INSERT INTO members (id, name, email, status) VALUES (?, ?, ?, ?)',
        [uuidv4(), app.name, app.email, 'active']
      )
    }
    
    // 记录审计日志
    await connection.execute(
      'INSERT INTO activity_logs (user_id, action, entity, entity_id) VALUES (?, ?, ?, ?)',
      [reviewerId, `application_${status}`, 'application', applicationId]
    )
  })
}
```

```javascript
import { getPool } from './pool.js'
import { logger } from '../utils/logger.js'

export function startPoolMonitoring() {
  const pool = getPool()
  
  setInterval(() => {
    const poolStatus = pool.status()
    
    logger.info('数据库连接池状态', {
      acquired: poolStatus.acquiredConnections,
      free: poolStatus.freeConnections,
      queued: poolStatus.queuedConnections,
      total: poolStatus.totalConnections
    })
    
    // 告警：连接池接近上限
    if (poolStatus.queuedConnections > 5) {
      logger.warn('数据库连接池排队警告', {
        queued: poolStatus.queuedConnections
      })
    }
  }, 60000)  // 每分钟检查
}
```

**Step 2：添加慢查询日志（`server/src/middleware/requestLogger.js`）**
```javascript
export function requestLogger(req, res, next) {
  const start = Date.now()
  
  res.on('finish', () => {
    const duration = Date.now() - start
    
    // 记录慢查询（> 1 秒）
    if (duration > 1000) {
      logger.warn('慢请求', {
        method: req.method,
        url: req.url,
        duration: `${duration}ms`,
        userAgent: req.get('User-Agent')
      })
    }
    
    // 正常日志
    logger.info('请求完成', {
      method: req.method,
      url: req.url,
      status: res.statusCode,
      duration: `${duration}ms`
    })
  })
  
  next()
}
```
